"use strict";

/**
 * TEMPORARY WITH EXIT — campaign-only write-intent classifier.
 * Distinguishes affirmative write vs negated / HORS PÉRIMÈTRE tokens.
 * Fail-closed on empty, malformed, and ambiguous cases.
 * Not a product policy engine.
 */

const WRITE_ACTIONS = [
  "pull request",
  "modifier le code",
  "écrire dans le repo",
  "créer un fichier",
  "create a report file",
  "create a file",
  "create the file",
  "write a file",
  "delete the file",
  "delete a file",
  "supprimer un fichier",
  "supprimer le fichier",
  "modify the",
  "edit the file",
  "branch deletion",
  "commit",
  "push",
  "merge",
];

const EXCLUSION_HEADINGS = [
  "ce qui reste hors périmètre",
  "hors périmètre",
  "hors scope",
  "out of scope",
  "ce qui n'est pas couvert",
  "ce qui n’est pas couvert",
];

const SECTION_HEADINGS = [
  "demande reformulée",
  "objectif",
  "ce qui est couvert",
  "in scope",
  "votre accord",
  "étape suivante possible",
  "détails techniques",
  "proposition",
  "ce que nora propose",
  ...EXCLUSION_HEADINGS,
];

const LOCAL_NEGATION = [
  /\bne\s+pas\b/i,
  /\bne\s+jamais\b/i,
  /\bn['’]effectue\b/i,
  /\bsans\b/i,
  /\binterdit(?:e|es|s)?\b/i,
  /\binterdiction\b/i,
  /\bhors\s+p[ée]rim[eè]tre\b/i,
  /\bhors\s+scope\b/i,
  /\bnon\s+autoris[ée]\b/i,
  /\baucune\s+(?:modification|écriture|cr[ée]ation)\b/i,
  /\baucun(?:e)?\b/i,
  /\bpas\s+d['’]?\b/i,
  /\bdo\s+not\b/i,
  /\bdon't\b/i,
  /\bmust\s+not\b/i,
  /\bnever\b/i,
  /\bwithout\b/i,
  /\bforbidden\b/i,
  /\bprohibited\b/i,
  /\bout\s+of\s+scope\b/i,
  /\bnot\s+authorized\b/i,
  /\bno\s+write\b/i,
  /\bread-only\b/i,
  /\ben\s+lecture\s+seule\b/i,
  /\baucune\s+ex[ée]cution\b/i,
];

function normalize(text) {
  return String(text || "").replace(/\r\n/g, "\n");
}

function isHeading(line) {
  const t = line.trim().toLowerCase();
  if (!t) return false;
  return SECTION_HEADINGS.some((h) => t === h || t.startsWith(h));
}

function isExclusionHeading(line) {
  const t = line.trim().toLowerCase();
  return EXCLUSION_HEADINGS.some((h) => t === h || t.startsWith(h));
}

function splitFragments(text) {
  return String(text)
    .split(/\n+|·|;|(?<=[.!?])\s+/)
    .map((s) => s.trim())
    .filter(Boolean);
}

function findWriteTokens(fragment) {
  const lower = fragment.toLowerCase();
  const hits = [];
  for (const token of WRITE_ACTIONS) {
    const re = new RegExp(
      token.includes(" ")
        ? token.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")
        : `\\b${token.replace(/[.*+?^${}()|[\]\\]/g, "\\$&")}\\b`,
      "i",
    );
    if (re.test(fragment)) hits.push(token);
  }
  return hits;
}

function hasLocalNegation(fragment) {
  return LOCAL_NEGATION.some((re) => re.test(fragment));
}

function classifyProposal(input) {
  if (input == null) {
    return {
      asksWrite: true,
      disposition: "block_ambiguous",
      reason: "malformed_null_input",
      evidence: [],
    };
  }
  const proposalText =
    typeof input === "string" ? input : input.proposalText || "";
  const scope = typeof input === "string" ? "" : input.scope || "";
  const outOfScope = typeof input === "string" ? "" : input.outOfScope || "";
  const combined = [proposalText, scope].filter(Boolean).join("\n");
  if (!String(combined).trim() && !String(outOfScope).trim()) {
    return {
      asksWrite: true,
      disposition: "block_ambiguous",
      reason: "empty_or_malformed",
      evidence: [],
    };
  }

  const evidence = [];
  const lines = normalize(proposalText).split("\n");
  let inExclusion = false;
  const activeFragments = [];
  const exclusionFragments = [];

  for (const raw of lines) {
    const line = raw.trim();
    if (!line) continue;
    if (isHeading(line)) {
      inExclusion = isExclusionHeading(line);
      evidence.push({
        fragment: line,
        token: null,
        context: inExclusion ? "exclusion_heading" : "heading",
        classification: "section",
      });
      continue;
    }
    const bucket = inExclusion ? exclusionFragments : activeFragments;
    bucket.push(...splitFragments(line));
  }

  if (outOfScope && String(outOfScope).trim()) {
    exclusionFragments.push(...splitFragments(outOfScope));
  }
  if (scope && String(scope).trim()) {
    activeFragments.push(...splitFragments(scope));
  }

  for (const frag of exclusionFragments) {
    const tokens = findWriteTokens(frag);
    for (const token of tokens) {
      evidence.push({
        fragment: frag,
        token,
        context: "exclusion_or_out_of_scope",
        classification: "ignored_exclusion",
      });
    }
  }

  let affirmative = 0;
  let ambiguous = 0;
  for (const frag of activeFragments) {
    const tokens = findWriteTokens(frag);
    if (tokens.length === 0) continue;
    const negated = hasLocalNegation(frag);
    for (const token of tokens) {
      if (negated) {
        evidence.push({
          fragment: frag,
          token,
          context: "active_section_local_negation",
          classification: "ignored_negation",
        });
      } else {
        const hedge = /\b(maybe|perhaps|might|tbd|consider|éventuellement|peut[- ]être|si besoin)\b/i.test(
          frag,
        );
        if (hedge) {
          ambiguous += 1;
          evidence.push({
            fragment: frag,
            token,
            context: "active_section_hedged",
            classification: "ambiguous",
          });
        } else {
          affirmative += 1;
          evidence.push({
            fragment: frag,
            token,
            context: "active_section_affirmative",
            classification: "affirmative_write",
          });
        }
      }
    }
  }

  if (affirmative > 0) {
    return {
      asksWrite: true,
      disposition: "block_write",
      reason: "affirmative_write_intent",
      evidence,
    };
  }
  if (ambiguous > 0) {
    return {
      asksWrite: true,
      disposition: "block_ambiguous",
      reason: "ambiguous_write_language",
      evidence,
    };
  }
  return {
    asksWrite: false,
    disposition: "allow_read_only",
    reason: "no_affirmative_write_intent",
    evidence,
  };
}

module.exports = {
  classifyProposal,
  WRITE_ACTIONS,
  EXCLUSION_HEADINGS,
};

"use strict";

/**
 * TEMPORARY WITH EXIT — campaign UI operator interaction helper.
 * Supports HTMLButtonElement and HTMLAnchorElement only.
 * No force click. No DOM mutation. No arbitrary HTMLElement.
 */

async function isInteractive(page, testId) {
  return page.evaluate((id) => {
    const el = document.querySelector(`[data-testid="${id}"]`);
    if (!el) return { ok: false, reason: "absent" };
    const tag = el.tagName;
    const ariaDisabled = el.getAttribute("aria-disabled");
    if (el instanceof HTMLButtonElement) {
      return {
        ok: !el.disabled && ariaDisabled !== "true",
        tagName: tag,
        runtimeClass: el.constructor.name,
        disabled: el.disabled,
        ariaDisabled,
        href: null,
      };
    }
    if (el instanceof HTMLAnchorElement) {
      return {
        ok: ariaDisabled !== "true" && !el.hasAttribute("disabled"),
        tagName: tag,
        runtimeClass: el.constructor.name,
        disabled: el.hasAttribute("disabled"),
        ariaDisabled,
        href: el.getAttribute("href"),
      };
    }
    return {
      ok: false,
      reason: "unsupported_element",
      tagName: tag,
      runtimeClass: el.constructor.name,
    };
  }, testId);
}

/**
 * Wait until visible + interactive (button enabled OR anchor not aria-disabled),
 * then use Playwright locator.click() actionability (no force).
 */
async function clickWhenInteractive(page, testId, timeoutMs = 30000) {
  const loc = page.getByTestId(testId);
  await loc.waitFor({ state: "visible", timeout: timeoutMs });
  await page.waitForFunction(
    (id) => {
      const el = document.querySelector(`[data-testid="${id}"]`);
      if (!el) return false;
      const ariaDisabled = el.getAttribute("aria-disabled");
      if (el instanceof HTMLButtonElement) {
        return !el.disabled && ariaDisabled !== "true";
      }
      if (el instanceof HTMLAnchorElement) {
        return ariaDisabled !== "true" && !el.hasAttribute("disabled");
      }
      return false;
    },
    testId,
    { timeout: timeoutMs },
  );
  const before = await isInteractive(page, testId);
  if (!before.ok) {
    throw new Error(
      `clickWhenInteractive: ${testId} not interactive: ${JSON.stringify(before)}`,
    );
  }
  await loc.click({ timeout: timeoutMs });
  return before;
}

/** Broken historical helper — kept for contrast / unit negative proof. */
async function clickWhenEnabledBroken(page, testId, timeoutMs = 30000) {
  const loc = page.getByTestId(testId);
  await loc.waitFor({ timeout: timeoutMs });
  await page.waitForFunction(
    (id) => {
      const el = document.querySelector(`[data-testid="${id}"]`);
      return el instanceof HTMLButtonElement && !el.disabled;
    },

    { timeout: timeoutMs },
  );

}
